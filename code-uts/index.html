Step 1: prepare samples (01_prepare_samples.py)¶

Maps labels to the five class names, fixes invalid geometries, drops polygons that are too small.

01_prepare_samples.py
"""Step 1 - check and clean your digitized polygons.

Reads data/samples.geojson, converts the label column to `kelas` with the five official
values (built_up, farmland, water, forest, mangrove), drops polygons that are too small
to contain pixels, and writes data/samples_clean.geojson.
"""
import sys

import geopandas as gpd

import config as C


def main():
    if not C.SAMPLES_RAW.exists():
        sys.exit(f"Put your polygon file at {C.SAMPLES_RAW}")

    gdf = gpd.read_file(C.SAMPLES_RAW)
    print(f"Read {len(gdf)} features, geometry types: {sorted(gdf.geom_type.unique())}")
    gdf = gdf[gdf.geom_type.isin(["Polygon", "MultiPolygon"])].copy()

    label_col = "kelas" if "kelas" in gdf.columns else "Type" if "Type" in gdf.columns else None
    if label_col is None:
        sys.exit(f"No 'kelas' or 'Type' column found. Columns: {list(gdf.columns)}")
    print(f"Label column used: {label_col}")

    raw = gdf[label_col].astype(str).str.strip().str.lower()
    unknown = sorted(set(raw) - set(C.LABEL_MAP))
    if unknown:
        sys.exit(f"Unknown label values {unknown}. Add them to LABEL_MAP in config.py or fix the file.")
    gdf["kelas"] = raw.map(C.LABEL_MAP)

    gdf = gdf[gdf.geometry.notna() & ~gdf.geometry.is_empty].copy()
    invalid = ~gdf.geometry.is_valid
    if invalid.any():
        print(f"Fixing {int(invalid.sum())} invalid geometries")
        gdf.loc[invalid, "geometry"] = gdf.loc[invalid, "geometry"].buffer(0)

    gdf["area_m2"] = gdf.to_crs(C.CRS).geometry.area.round(0)
    too_small = gdf["area_m2"] < C.MIN_POLYGON_AREA_M2
    too_large = gdf["area_m2"] > C.WARN_POLYGON_AREA_M2
    print(f"Dropping {int(too_small.sum())} polygons smaller than {C.MIN_POLYGON_AREA_M2} m2")
    print(f"WARNING: {int(too_large.sum())} polygons larger than {C.WARN_POLYGON_AREA_M2 / 1e6:.0f} km2 "
          f"(kept, but they may cover mixed land cover - consider redrawing)")
    gdf = gdf[~too_small].copy()

    gdf["class_id"] = gdf["kelas"].map(C.CLASS_ID)
    gdf = gdf.sort_values("kelas").reset_index(drop=True)
    gdf["poly_id"] = gdf.groupby("kelas").cumcount().add(1).map("{:03d}".format)
    gdf["poly_id"] = gdf["kelas"] + "_" + gdf["poly_id"]

    out = gdf[["poly_id", "kelas", "class_id", "area_m2", "geometry"]].to_crs(4326)
    C.DATA.mkdir(exist_ok=True)
    out.to_file(C.SAMPLES_CLEAN, driver="GeoJSON")

    print("\nPolygons per class:")
    print(out.groupby("kelas").agg(polygons=("poly_id", "count"),
                                   median_area_m2=("area_m2", "median"),
                                   max_area_m2=("area_m2", "max")).reindex(C.CLASSES))
    print(f"\nSaved {len(out)} polygons -> {C.SAMPLES_CLEAN}")


if __name__ == "__main__":
    main()

Step 2: download Sentinel-2 (02_download_sentinel2.py)¶

Cloud-masked median composite of bands B04, B8A, B11 for 2025, downloaded tile by tile from Copernicus openEO.

02_download_sentinel2.py
"""Step 2 - download a Sentinel-2 L2A composite (bands B04, B8A, B11) from Copernicus openEO.

  python 02_download_sentinel2.py --test      small 10x10 km test around your first farmland polygon
  python 02_download_sentinel2.py --dry-run   list the tiles that would be downloaded, then stop
  python 02_download_sentinel2.py --samples-only   only tiles that contain your polygons (enough for training)
  python 02_download_sentinel2.py             download all tiles (resumes: existing files are skipped)

For every tile the server: filters Sentinel-2 L2A for 2025, masks cloud / cloud shadow / cirrus /
snow with the SCL band, takes the per-pixel median over time, and returns a 3-band GeoTIFF
(B04, B8A, B11) at 20 m in EPSG:32749.
A browser window opens the first time so you can log in to your Copernicus account.
"""
import argparse
import sys
import time

import geopandas as gpd
import openeo
from shapely.geometry import box

import config as C
from geo_utils import load_boundary_utm
from lulc_utils import tile_grid


def build_cube(conn, west, south, east, north):
    cube = conn.load_collection(
        "SENTINEL2_L2A",
        spatial_extent={"west": west, "south": south, "east": east, "north": north, "crs": C.CRS},
        temporal_extent=[C.DATE_START, C.DATE_END],
        bands=C.BANDS + ["SCL"],
        max_cloud_cover=C.MAX_CLOUD,
    )
    scl = cube.band("SCL")
    # SCL classes: 3 cloud shadow, 8 cloud medium, 9 cloud high, 10 thin cirrus, 11 snow/ice
    bad = (scl == 3) | (scl == 8) | (scl == 9) | (scl == 10) | (scl == 11)
    cube = cube.filter_bands(C.BANDS).mask(bad)
    cube = cube.reduce_dimension(dimension="t", reducer="median")
    cube = cube.resample_spatial(resolution=C.RESOLUTION, projection=C.EPSG)
    return cube


def run_job(conn, extent, out_path, title):
    west, south, east, north = extent
    cube = build_cube(conn, west, south, east, north)
    job = cube.create_job(title=title, out_format="GTiff")
    job.start_and_wait()
    out_path.parent.mkdir(parents=True, exist_ok=True)
    job.get_results().download_file(out_path)


def test_extent():
    gdf = gpd.read_file(C.SAMPLES_CLEAN).to_crs(C.CRS)
    farm = gdf[gdf["kelas"] == "farmland"]
    if farm.empty:
        sys.exit("No farmland polygon found in samples_clean.geojson (run step 1 first).")
    c = farm.iloc[0].geometry.centroid
    half = 5000
    snap = lambda v: round(v / C.RESOLUTION) * C.RESOLUTION
    return snap(c.x - half), snap(c.y - half), snap(c.x + half), snap(c.y + half)


def all_tiles(samples_only=False):
    boundary = load_boundary_utm()
    minx, miny, maxx, maxy = boundary.bounds
    polys = gpd.read_file(C.SAMPLES_CLEAN).to_crs(C.CRS) if samples_only else None
    tiles = []
    for name, x0, y0, x1, y1 in tile_grid(minx, miny, maxx, maxy, C.TILE_SIZE_M):
        b = box(x0, y0, x1, y1)
        if not b.intersects(boundary):
            continue
        if samples_only and not polys.intersects(b).any():
            continue
        tiles.append((name, (x0, y0, x1, y1)))
    return tiles


def main():
    ap = argparse.ArgumentParser()
    ap.add_argument("--test", action="store_true")
    ap.add_argument("--dry-run", action="store_true")
    ap.add_argument("--samples-only", action="store_true",
                    help="only tiles that contain sample polygons")
    args = ap.parse_args()

    if args.test:
        extent = test_extent()
        print("Test extent (UTM 49S, metres):", extent)
        conn = openeo.connect("openeo.dataspace.copernicus.eu").authenticate_oidc()
        out = C.TEST_DIR / "test_area.tif"
        t0 = time.time()
        run_job(conn, extent, out, "uts-test-area")
        print(f"Saved {out} in {time.time() - t0:.0f} s. Now open it and check it has 3 bands.")
        print("Check your credit usage in the Copernicus dashboard before running the full download.")
        return

    tiles = all_tiles(args.samples_only)
    print(f"{len(tiles)} tiles of {C.TILE_SIZE_M // 1000} km x {C.TILE_SIZE_M // 1000} km")
    if args.dry_run:
        for name, ext in tiles:
            print(" ", name, ext)
        return

    conn = openeo.connect("openeo.dataspace.copernicus.eu").authenticate_oidc()
    failed = []
    for i, (name, (x0, y0, x1, y1)) in enumerate(tiles, 1):
        out = C.TILES_DIR / f"{name}.tif"
        if out.exists():
            print(f"[{i}/{len(tiles)}] {name}: already downloaded, skipping")
            continue
        print(f"[{i}/{len(tiles)}] {name}: requesting ...")
        try:
            t0 = time.time()
            run_job(conn, (x0, y0, x1, y1), out, f"uts-{name}")
            print(f"    done in {time.time() - t0:.0f} s")
        except Exception as e:  # keep going; rerun the script to retry failed tiles
            print(f"    FAILED: {e}")
            failed.append(name)
    if failed:
        print("\nFailed tiles (rerun this script to retry):", failed)
    else:
        print("\nAll tiles downloaded.")


if __name__ == "__main__":
    main()

Step 3: extract pixels, train, evaluate (03_extract_and_train.py)¶

Reads pixel values under the polygons, trains the Random Forest, evaluates on test polygons.

03_extract_and_train.py
"""Step 3 - read pixel values under your polygons, train the classifier, evaluate it.

Model: Random Forest on B04, B8A, B11 (+ NDVI and NDMI computed from them).
The train/test split is made by POLYGON, so pixels of one polygon never appear in both sets.
"""
import json
import sys

import geopandas as gpd
import joblib
import numpy as np
import pandas as pd
import rasterio
from rasterio.mask import mask as rio_mask
from shapely.geometry import box, mapping
from sklearn.ensemble import RandomForestClassifier

import config as C
from lulc_utils import evaluate, make_features, split_polygons, valid_pixels


def extract_pixels():
    tiles = sorted(C.TILES_DIR.glob("*.tif"))
    if not tiles:
        sys.exit(f"No tiles in {C.TILES_DIR}. Run step 2 first.")
    polys = gpd.read_file(C.SAMPLES_CLEAN).to_crs(C.CRS)
    rng = np.random.default_rng(C.SEED)
    rows = []
    for tif in tiles:
        with rasterio.open(tif) as src:
            if src.count < 3:
                sys.exit(f"{tif.name} has {src.count} bands, expected 3 (B04, B8A, B11).")
            hit = polys[polys.intersects(box(*src.bounds))]
            for _, p in hit.iterrows():
                try:
                    arr, _ = rio_mask(src, [mapping(p.geometry)], crop=True, filled=False)
                except ValueError:      # polygon does not overlap this raster
                    continue
                bands = arr[:3].reshape(3, -1)
                inside = ~np.ma.getmaskarray(bands).any(axis=0)
                vals = np.ma.getdata(bands).astype("float32")
                if src.nodata is not None and not np.isnan(src.nodata):
                    inside &= ~np.any(vals == src.nodata, axis=0)
                ok = inside & valid_pixels(vals)
                idx = np.flatnonzero(ok)
                if len(idx) == 0:
                    continue
                if len(idx) > C.MAX_PIXELS_PER_POLYGON:
                    idx = rng.choice(idx, C.MAX_PIXELS_PER_POLYGON, replace=False)
                df = make_features(vals[0, idx], vals[1, idx], vals[2, idx])
                df.insert(0, "kelas", p["kelas"])
                df.insert(0, "poly_id", p["poly_id"])
                rows.append(df)
    if not rows:
        sys.exit("No pixels extracted: polygons do not overlap the downloaded tiles.")
    return pd.concat(rows, ignore_index=True)


def main():
    pix = extract_pixels()
    pix.to_csv(C.PIXELS_CSV, index=False)
    used = pix.groupby("kelas")["poly_id"].nunique().reindex(C.CLASSES).fillna(0).astype(int)
    print("Polygons with usable pixels per class:\n", used.to_string())
    print("Pixels per class:\n", pix["kelas"].value_counts().reindex(C.CLASSES).fillna(0).astype(int).to_string())

    poly_df = pix[["poly_id", "kelas"]].drop_duplicates()
    train_ids, test_ids = split_polygons(poly_df, C.TEST_SIZE, C.SEED)
    tr = pix[pix["poly_id"].isin(train_ids)]
    te = pix[pix["poly_id"].isin(test_ids)]
    print(f"\nTrain: {len(train_ids)} polygons / {len(tr)} pixels   Test: {len(test_ids)} polygons / {len(te)} pixels")

    model = RandomForestClassifier(**C.RF_PARAMS)
    model.fit(tr[C.FEATURES], tr["kelas"])
    pred = model.predict(te[C.FEATURES])

    present = [c for c in C.CLASSES if c in set(te["kelas"]) | set(pred)]
    res = evaluate(te["kelas"], pred, present)
    res["n_polygons_train"] = len(train_ids)
    res["n_polygons_test"] = len(test_ids)
    res["n_pixels_train"] = int(len(tr))
    res["n_pixels_test"] = int(len(te))
    res["polygons_per_class"] = {k: int(v) for k, v in used.items()}
    res["pixels_per_class"] = {k: int(v) for k, v in pix["kelas"].value_counts().items()}
    res["feature_importance"] = dict(zip(C.FEATURES, map(float, model.feature_importances_)))
    res["period"] = f"{C.DATE_START} to {C.DATE_END}"

    print(f"\nOverall accuracy: {res['accuracy']:.4f}   Cohen's kappa: {res['kappa']:.4f}")
    print("\nPer-class F1 (lowest first - these classes need more / better polygons):")
    f1 = {c: res["per_class"][c]["f1-score"] for c in present}
    for c, v in sorted(f1.items(), key=lambda kv: kv[1]):
        print(f"  {c:10s} {v:.3f}")
    print("\nConfusion matrix (rows = true, columns = predicted):")
    print(pd.DataFrame(res["confusion_matrix"], index=present, columns=present).to_string())

    sig = pix.groupby("kelas")[["B04", "B8A", "B11"]].mean().reindex(C.CLASSES).dropna()
    sig.to_csv(C.SIGNATURE_CSV)
    joblib.dump({"model": model, "features": C.FEATURES}, C.MODEL_PATH)
    with open(C.METRICS_JSON, "w") as f:
        json.dump(res, f, indent=2)
    print(f"\nSaved {C.MODEL_PATH.name}, {C.METRICS_JSON.name}, {C.SIGNATURE_CSV.name}, {C.PIXELS_CSV.name} in data/")


if __name__ == "__main__":
    main()

Step 4: predict and create GeoJSON layers (04_predict_and_vectorize.py)¶

Classifies every tile and converts each class into a GeoJSON layer.

04_predict_and_vectorize.py
"""Step 4 - classify every tile, then digitize the result into one GeoJSON layer per class.

For each Sentinel-2 tile: predict a class map (GeoTIFF, 1-5, 0 = no data / outside East Java),
count the area per class, and convert each class to polygons (GeoJSON, EPSG:4326) for the web map.
Polygons are generalized (coarser grid, small blobs removed, simplified) so the files stay small
enough for Leaflet - see VECTOR_FACTOR, SIEVE_MIN_PIXELS, MIN_AREA_HA and SIMPLIFY_M in config.py.
"""
import json
import sys

import geopandas as gpd
import joblib
import numpy as np
import rasterio
from rasterio.features import geometry_mask, shapes, sieve
from shapely.geometry import box, shape

import config as C
from geo_utils import load_boundary_utm
from lulc_utils import make_features, valid_pixels


def predict_tile(model, features, src, boundary):
    data = src.read([1, 2, 3]).astype("float32")             # (3, h, w)
    h, w = data.shape[1:]
    flat = data.reshape(3, -1)
    ok = valid_pixels(flat)
    inside = geometry_mask([boundary], out_shape=(h, w), transform=src.transform, invert=True).ravel()
    ok &= inside
    out = np.zeros(h * w, dtype="uint8")
    idx = np.flatnonzero(ok)
    for start in range(0, len(idx), 500_000):                # predict in chunks to save memory
        sl = idx[start:start + 500_000]
        X = make_features(flat[0, sl], flat[1, sl], flat[2, sl])[features]
        out[sl] = [C.CLASS_ID[k] for k in model.predict(X)]
    return out.reshape(h, w)


def class_polygons(class_map, transform, class_id):
    """Binary class layer -> list of shapely polygons in the raster CRS (metres)."""
    f = C.VECTOR_FACTOR
    h, w = class_map.shape
    hh, ww = h // f * f, w // f * f
    binary = (class_map[:hh, :ww] == class_id).astype("uint8")
    if f > 1:                                                # majority vote per f x f block
        blocks = binary.reshape(hh // f, f, ww // f, f).sum(axis=(1, 3), dtype="uint8")
        binary = (blocks * 2 >= f * f).astype("uint8")
    t = transform * transform.scale(f, f)
    if not binary.any():
        return []
    binary = sieve(binary, size=C.SIEVE_MIN_PIXELS, connectivity=4)
    polys = []
    min_area = C.MIN_AREA_HA * 10_000
    for geom, val in shapes(binary, mask=binary.astype(bool), transform=t, connectivity=4):
        g = shape(geom)
        if g.area < min_area:
            continue
        polys.append(g.simplify(C.SIMPLIFY_M, preserve_topology=True))
    return polys


def round_coords(obj, nd=5):
    if isinstance(obj, (list, tuple)):
        if obj and isinstance(obj[0], (int, float)):
            return [round(v, nd) for v in obj]
        return [round_coords(o, nd) for o in obj]
    return obj


def main():
    tiles = sorted(C.TILES_DIR.glob("*.tif"))
    if not tiles:
        sys.exit(f"No tiles in {C.TILES_DIR}. Run step 2 first.")
    bundle = joblib.load(C.MODEL_PATH)
    model, features = bundle["model"], bundle["features"]
    boundary = load_boundary_utm()

    C.CLASS_TILES_DIR.mkdir(parents=True, exist_ok=True)
    C.GEOJSON_DIR.mkdir(parents=True, exist_ok=True)
    pixel_counts = {k: 0 for k in C.CLASSES}
    layer_polys = {k: [] for k in C.CLASSES}
    res = C.RESOLUTION

    for i, tif in enumerate(tiles, 1):
        print(f"[{i}/{len(tiles)}] {tif.name}")
        with rasterio.open(tif) as src:
            if not box(*src.bounds).intersects(boundary):
                continue
            cmap = predict_tile(model, features, src, boundary)
            profile = src.profile.copy()
            profile.update(count=1, dtype="uint8", nodata=0, compress="lzw")
            with rasterio.open(C.CLASS_TILES_DIR / f"{tif.stem}_class.tif", "w", **profile) as dst:
                dst.write(cmap, 1)
            for k in C.CLASSES:
                cid = C.CLASS_ID[k]
                pixel_counts[k] += int((cmap == cid).sum())
                layer_polys[k] += class_polygons(cmap, src.transform, cid)

    total = sum(pixel_counts.values())
    stats = {k: {"pixels": v,
                 "area_km2": round(v * res * res / 1e6, 2),
                 "percent": round(100 * v / total, 2) if total else 0.0}
             for k, v in pixel_counts.items()}
    with open(C.STATS_JSON, "w") as f:
        json.dump({"resolution_m": res, "classes": stats}, f, indent=2)

    for k in C.CLASSES:
        gdf = gpd.GeoDataFrame({"kelas": k, "area_ha": [round(g.area / 1e4, 2) for g in layer_polys[k]]},
                               geometry=layer_polys[k], crs=C.CRS).to_crs(4326)
        gj = json.loads(gdf.to_json(drop_id=True))
        for feat in gj["features"]:
            feat["geometry"]["coordinates"] = round_coords(feat["geometry"]["coordinates"])

"""Helpers that need only numpy / pandas / scikit-learn (no GIS libraries)."""
import math

import numpy as np
import pandas as pd

import config as C


def make_features(b04, b8a, b11):
    """Build the model feature table from three 1-D arrays of reflectance values."""
    b04 = np.asarray(b04, dtype="float32")
    b8a = np.asarray(b8a, dtype="float32")
    b11 = np.asarray(b11, dtype="float32")
    eps = 1e-6
    df = pd.DataFrame({
        "B04": b04,
        "B8A": b8a,
        "B11": b11,
        "NDVI": (b8a - b04) / (b8a + b04 + eps),   # vegetation
        "NDMI": (b8a - b11) / (b8a + b11 + eps),   # moisture (NIR vs SWIR)
    })
    return df[C.FEATURES]


def valid_pixels(bands):
    """bands: array (3, n). True where the pixel holds usable values."""
    ok = np.all(np.isfinite(bands), axis=0)
    ok &= np.any(bands != 0, axis=0)
    return ok


def split_polygons(poly_df, test_size, seed):
    """Stratified split by polygon. poly_df needs columns poly_id and kelas."""
    rng = np.random.default_rng(seed)
    train, test = [], []
    for _, g in poly_df.groupby("kelas"):
        ids = g["poly_id"].to_numpy().copy()
        rng.shuffle(ids)
        n_test = max(1, int(round(len(ids) * test_size))) if len(ids) > 1 else 0
        test += list(ids[:n_test])
        train += list(ids[n_test:])
    return set(train), set(test)


def tile_grid(minx, miny, maxx, maxy, size):
    """Yield (name, x0, y0, x1, y1) for a grid aligned to multiples of `size`."""
    x_start = math.floor(minx / size) * size
    y_start = math.floor(miny / size) * size
    x = x_start
    while x < maxx:
        y = y_start
        while y < maxy:
            yield f"tile_{int(x // 1000)}_{int(y // 1000)}", x, y, x + size, y + size
            y += size
        x += size


def evaluate(y_true, y_pred, labels):
    from sklearn.metrics import (accuracy_score, classification_report,
                                 cohen_kappa_score, confusion_matrix)
    return {
        "accuracy": float(accuracy_score(y_true, y_pred)),
        "kappa": float(cohen_kappa_score(y_true, y_pred)),
        "labels": list(labels),
        "confusion_matrix": confusion_matrix(y_true, y_pred, labels=labels).tolist(),
        "per_class": classification_report(y_true, y_pred, labels=labels,
                                           output_dict=True, zero_division=0),
    }

"""Helpers that need geopandas / shapely."""
import geopandas as gpd
from shapely.ops import unary_union

import config as C


def load_boundary_utm():
    """East Java boundary in EPSG:32749 as one shapely geometry (far islands removed if configured)."""
    gdf = gpd.read_file(C.BOUNDARY)
    geom = unary_union(list(gdf.to_crs(C.CRS).geometry))
    parts = list(geom.geoms) if geom.geom_type == "MultiPolygon" else [geom]
    if C.EXCLUDE_FAR_ISLANDS:
        kept = []
        for p in parts:
            c = gpd.GeoSeries([p], crs=C.CRS).to_crs(4326).iloc[0].centroid
            if c.x > C.MAX_LON or c.y > C.MAX_LAT:
                continue
            kept.append(p)
        parts = kept
    geom = unary_union(parts)
    if C.BOUNDARY_BUFFER_M:
        geom = geom.buffer(C.BOUNDARY_BUFFER_M)
    return geom
